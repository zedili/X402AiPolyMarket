# 模块3：交易模块 - 验收清单

## 📋 功能清单

### 1. 订单管理 ✅

#### 1.1 创建订单
- **接口**: `POST /api/v1/trade/order`
- **功能**: 创建买入或卖出订单
- **验证点**:
  - [ ] 验证市场是否存在且可交易
  - [ ] 验证市场状态（必须是进行中）
  - [ ] 验证市场是否已结束
  - [ ] 正确计算订单总价值和手续费（2%）
  - [ ] 订单成功创建并返回订单信息
  - [ ] 订单信息缓存到Redis（10分钟）
  - [ ] 需要JWT认证

#### 1.2 订单列表
- **接口**: `GET /api/v1/trade/orders`
- **功能**: 获取用户的订单列表
- **验证点**:
  - [ ] 支持分页（page, page_size）
  - [ ] 支持按市场ID筛选
  - [ ] 支持按订单状态筛选
  - [ ] 支持按订单类型筛选
  - [ ] 返回订单总数和订单列表
  - [ ] 包含市场问题信息
  - [ ] 订单按创建时间倒序排列
  - [ ] 需要JWT认证

#### 1.3 订单详情
- **接口**: `GET /api/v1/trade/order/:id`
- **功能**: 获取订单详细信息
- **验证点**:
  - [ ] 验证订单是否存在
  - [ ] 验证订单属于当前用户
  - [ ] 返回完整订单信息
  - [ ] 包含关联的交易记录
  - [ ] 需要JWT认证

#### 1.4 取消订单
- **接口**: `POST /api/v1/trade/order/:id/cancel`
- **功能**: 取消未成交或部分成交的订单
- **验证点**:
  - [ ] 验证订单是否存在
  - [ ] 验证订单属于当前用户
  - [ ] 不能取消已完全成交的订单
  - [ ] 不能取消已取消的订单
  - [ ] 更新订单状态为已取消
  - [ ] 记录取消时间
  - [ ] 需要JWT认证

### 2. 交易历史 ✅

```bash
#!/bin/bash

# 配置
API_URL="http://localhost:8888/api/v1"
TOKEN="YOUR_JWT_TOKEN_HERE"

# 颜色输出
GREEN='\033[0;32m'
RED='\033[0;31m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# 测试计数
TOTAL_TESTS=0
PASSED_TESTS=0
FAILED_TESTS=0

# 测试函数
test_api() {
    local test_name=$1
    local method=$2
    local endpoint=$3
    local data=$4
    local expected_code=$5

    TOTAL_TESTS=$((TOTAL_TESTS + 1))
    echo -e "${YELLOW}Testing: $test_name${NC}"

    if [ -z "$data" ]; then
        response=$(curl -s -w "\n%{http_code}" -X $method "$API_URL$endpoint" \
            -H "Authorization: Bearer $TOKEN")
    else
        response=$(curl -s -w "\n%{http_code}" -X $method "$API_URL$endpoint" \
            -H "Authorization: Bearer $TOKEN" \
            -H "Content-Type: application/json" \
            -d "$data")
    fi

    http_code=$(echo "$response" | tail -n1)
    body=$(echo "$response" | sed '$d')

    if [ "$http_code" -eq "$expected_code" ]; then
        echo -e "${GREEN}✓ PASSED${NC} (HTTP $http_code)"
        PASSED_TESTS=$((PASSED_TESTS + 1))
    else
        echo -e "${RED}✗ FAILED${NC} (Expected HTTP $expected_code, got $http_code)"
        FAILED_TESTS=$((FAILED_TESTS + 1))
    fi

    echo "Response: $body"
    echo ""
}

# 开始测试
echo "========================================="
echo "交易模块API测试"
echo "========================================="
echo ""

# 1. 订单管理测试
```

### PowerShell测试脚本

```powershell
# 配置
$API_URL = "http://localhost:8888/api/v1"
$TOKEN = "YOUR_JWT_TOKEN_HERE"

# 测试计数
$TotalTests = 0
$PassedTests = 0
$FailedTests = 0

# 测试函数
function Test-API {
    param(
        [string]$TestName,
        [string]$Method,
        [string]$Endpoint,
        [string]$Data = $null,
        [int]$ExpectedCode = 200
    )

    $script:TotalTests++
    Write-Host "Testing: $TestName" -ForegroundColor Yellow

    $headers = @{
        "Authorization" = "Bearer $TOKEN"
        "Content-Type" = "application/json"
    }

    try {
        if ($Data) {
            $response = Invoke-WebRequest -Uri "$API_URL$Endpoint" `
                -Method $Method -Headers $headers -Body $Data `
                -UseBasicParsing -ErrorAction Stop
        } else {
            $response = Invoke-WebRequest -Uri "$API_URL$Endpoint" `
                -Method $Method -Headers $headers `
                -UseBasicParsing -ErrorAction Stop
        }

        if ($response.StatusCode -eq $ExpectedCode) {
            Write-Host "✓ PASSED (HTTP $($response.StatusCode))" -ForegroundColor Green
            $script:PassedTests++
        } else {
            Write-Host "✗ FAILED (Expected HTTP $ExpectedCode, got $($response.StatusCode))" -ForegroundColor Red
            $script:FailedTests++
        }

        Write-Host "Response: $($response.Content)"
    } catch {
        $statusCode = $_.Exception.Response.StatusCode.value__
        if ($statusCode -eq $ExpectedCode) {
            Write-Host "✓ PASSED (HTTP $statusCode)" -ForegroundColor Green
            $script:PassedTests++
        } else {
            Write-Host "✗ FAILED (Expected HTTP $ExpectedCode, got $statusCode)" -ForegroundColor Red
            $script:FailedTests++
        }
    }

    Write-Host ""
}

# 开始测试
Write-Host "=========================================" -ForegroundColor Cyan
Write-Host "交易模块API测试" -ForegroundColor Cyan
Write-Host "=========================================" -ForegroundColor Cyan

Test-API -TestName "查询订单详情" -Method "GET" -Endpoint "/trade/order/1"

Test-API -TestName "取消订单" -Method "POST" -Endpoint "/trade/order/1/cancel"

# 2. 交易历史测试
Write-Host "=== 交易历史测试 ===" -ForegroundColor Cyan
Test-API -TestName "查询交易历史" -Method "GET" -Endpoint "/trade/history?page=1&page_size=20"

# 3. 持仓管理测试
Write-Host "=== 持仓管理测试 ===" -ForegroundColor Cyan
Test-API -TestName "查询持仓列表" -Method "GET" -Endpoint "/trade/positions"

Test-API -TestName "查询持仓详情" -Method "GET" -Endpoint "/trade/position/1"

# 4. 统计测试
Write-Host "=== 统计测试 ===" -ForegroundColor Cyan
Test-API -TestName "查询交易统计" -Method "GET" -Endpoint "/trade/stats"

# 5. 错误场景测试
Write-Host "=== 错误场景测试 ===" -ForegroundColor Cyan

    -Data '{"market_id":99999,"order_type":0,"position":1,"amount":100,"price":67.5}' `
    -ExpectedCode 404

Test-API -TestName "订单不存在" -Method "GET" -Endpoint "/trade/order/99999" `
    -ExpectedCode 404

Write-Host "测试总结" -ForegroundColor Cyan
Write-Host "=========================================" -ForegroundColor Cyan
```
